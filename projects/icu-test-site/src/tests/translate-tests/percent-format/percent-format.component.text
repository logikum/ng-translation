title: Percent Formatting

basic_f: Basic percent format
basic_l: { shortPercent, number, ::percent }
basic_s: { shortPercent, number, ::percent }

minid_f: Minimum integer digits
minid_l: { shortPercent, number, ::percent integer-width/*000}
minid_s: { shortPercent, number, ::percent 000 }

minfd_f: Minimum fraction digits
minfd_l: ~
minfd_s: { shortPercent, number, ::percent .000*}

maxfd_f: Maximum fraction digits
maxfd_l: ~
maxfd_s: { shortPercent, number, ::percent .#}

minsd_f: Minimum significant digits
minsd_l: ~
minsd_s: { longPercent, number, ::percent .###/@@@*}

maxsd_f: Maximum significant digits
maxsd_l: ~
maxsd_s: { longPercent, number, ::percent @##}

mixed_f: With several options
mixed_l: { shortPercent, number, ::percent integer-width/*000 .00000*}
mixed_s: { shortPercent, number, ::percent 000 .00000* }

usegrp_f: Without grouping
usegrp_l: { longPercent, number, ::percent group-off}
usegrp_s: { longPercent, number, ::percent  ,_ }
