title: Date and Time Formatting

default_f: Default date format
default_l: { now, date } { now, time }
default_s: { now, date } { now, time }

short_f: Short date and time
short_l: { now, date, short } { now, time, short }
short_s: { now, date, short } { now, time, short }

medium_f: Medium date and time
medium_l: { now, date, medium } { now, time, medium }
medium_s: { now, date, medium } { now, time, medium }

long_f: Long date and time
long_l: { now, date, long } { now, time, long }
long_s: { now, date, long } { now, time, long }

full_f: Full date and time
full_l: { now, date, full } { now, time, full }
full_s: { now, date, full } { now, time, full }

fractional_f: Fractional second digits
fractional_l: { now, date, medium } { now, time, ::HHmmssSSS }
fractional_s: { now, date, medium } { now, time, ::HHmmssSSS }
